cmake_minimum_required(VERSION 3.15)
project(CarrierEnvelope LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
 message(FATAL_ERROR "The qualified envelope profile requires GCC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-frounding-math -ffp-contract=off -fno-fast-math)
find_package(Eigen3 REQUIRED)
find_package(spdlog REQUIRED)
find_package(CLI11 REQUIRED)
set(FAST_ENVELOPE_WITH_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(FAST_ENVELOPE_ENABLE_TBB OFF CACHE BOOL "" FORCE)
set(FAST_ENVELOPE_WITH_GEOGRAM_PREDICATES OFF CACHE BOOL "" FORCE)
set(FAST_ENVELOPE_WITH_GEOGRAM_PSM_PREDICATES ON CACHE BOOL "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_IPRED /input/indirect-predicates CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_SIMDE /input/simde CACHE PATH "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
add_subdirectory(/input/fast-envelope fast-envelope)
add_executable(check-envelope check.cpp)
target_link_libraries(check-envelope PRIVATE FastEnvelope)
