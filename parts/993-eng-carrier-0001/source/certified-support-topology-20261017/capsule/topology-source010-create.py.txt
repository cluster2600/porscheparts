def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda:stream.read(1048576),b''):h.update(block)
    return h.hexdigest()


def canonical(value):return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()


def path(name):
    p=Path(name)
    if p.is_absolute() or '..' in p.parts:raise ValueError('Only workspace-relative retained paths are allowed')
    return ROOT/p


def fresh(name):
    p=path(name)
    if p.exists() or p.is_symlink():raise ValueError('The candidate destination must be fresh')
    return p


def private(name,source):
    spec=importlib.util.spec_from_file_location(name,source)
    module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module)
    return module


def verify(plan):
    if (plan['schema']!='source010-certified-flip-caller-1' or plan['source_index']!=10 or
        plan['runtime_expected_sha256']!=RUNTIME_SHA or plan['total_wall_budget_seconds']!=300 or
        plan['threads']!=1 or plan['workers']!=0 or plan['nice']!=10 or
        plan['heights_mm']!=[(i+.5)*.06 for i in range(1034)]):
        raise ValueError('The fixed caller scope, calendar or runtime differs')
    if plan['files_sha256'].get(str(Path(__file__).resolve().relative_to(ROOT)))!=sha(__file__):
        raise ValueError('The caller source is not pinned')
    for role,digest in REQUIRED.items():
        if plan['files_sha256'].get(plan[role])!=digest:raise ValueError('A required retained identity differs: '+role)
    for name,digest in plan['files_sha256'].items():
        if sha(path(name))!=digest:raise ValueError('Changed retained input: '+name)
    fresh(plan['output_directory'])
    if json.loads((HERE/'plan.json').read_text())!=plan:raise ValueError('The caller plan changed after parsing')
    return dict(status='prepared_not_executed',plan_sha256=sha(HERE/'plan.json'),
                files_sha256=plan['files_sha256'],declared_layers=1034,workers=0,
                no_AM_imported=True,no_candidate_created=True,global_admission=False)


def bind_trace_groups(groups,components,trace,arrays):
    """Bind the measured per-chart error authority; introduce no new epsilon."""
    groups=deepcopy(groups);skins={g['id']:g for g in groups if g['kind']=='skin'}
    if {row['owner'] for row in trace['rows']}!=set(skins):raise ValueError('The trace does not cover exactly the SKIN owners')
    owners={}
    for row in trace['rows']:
        g=skins[row['owner']];ci=row['skin_component_index'];proof=row['trace_proof']
        if (g['face_range']!=row['global_face_range'] or g['mapping_t_xy_mm']!=row['mapping_t_xy_mm'] or
            row['remapping_receipt']!=components[ci]['remapping'] or
            proof['exact_zero_area_cleanup']!=row['remapping_receipt']['exact_zero_area_cleanup'] or
            not all(proof.get(k) is True for k in ('forward_XYZ_bytes_identical','raw_to_kept_faces_identical',
                'parameter_vertices_in_original_call_order','all_retained_faces_have_one_affine_branch',
                'every_actual_error_within_unchanged_original_gamma8_bound'))):
            raise ValueError('The original remapping/cleanup/encoding authority differs')
        bound=proof['maximum_exact_mapping_error_squared_mm2']
        if not isinstance(bound,str) or Q(bound)<0:raise ValueError('The measured trace bound must be an exact nonnegative ratio')
        g['trace_maximum_error_squared_mm2']=bound
        expected=list(range(row['global_vertex_offset'],row['global_vertex_offset']+proof['vertices']))
        actual=[int(i) for i,c in zip(arrays['global_vertex_ids'],arrays['skin_component_index']) if int(c)==ci]
        if actual!=expected:raise ValueError('The actual trace global ID block differs')
        owners[ci]=g['id']
    sizes={len(value) for value in arrays.values()}
    if len(sizes)!=1 or len(set(map(int,arrays['global_vertex_ids'])))!=len(arrays['global_vertex_ids']):
        raise ValueError('The actual trace array counts or unique IDs differ')
    parameters={}
    for i,tz,branch,ci in zip(arrays['global_vertex_ids'],arrays['vertex_parameters_tz'],arrays['vertex_mapping_branch'],arrays['skin_component_index']):
        if len(tz)!=2 or not all(math.isfinite(float(v)) for v in tz) or int(ci) not in owners:
            raise ValueError('Invalid original trace parameters')
        parameters[int(i)]=dict(owner=owners[int(ci)],tz=tuple(Q(float(v)) for v in tz),branch=int(branch))
    return groups,parameters


def verify_census_header(census,census_plan,reference):
    if (census.get('status')!='diagnostic_calendar_completed_no_global_admission' or
        census.get('declared_layers')!=1034 or census.get('completed_layers')!=1034 or
        census.get('collapsed_cuts')!=987 or census.get('runtime_unchanged') is not True or
        census.get('files_unchanged') is not True or census.get('worker_group_closed') is not True or
        census.get('fabrication_vectors_exported') is not False or
        census.get('files_sha256',{}).get('layers.jsonl')!=REQUIRED['census'] or
        census.get('plan_sha256')!=REQUIRED['census_plan'] or
        census_plan.get('surface')!=reference or
        census_plan.get('files_sha256',{}).get(reference)!=REQUIRED['reference']):
        raise ValueError('The retained full reference census is not complete and input-bound')

def bind_native_operations(packet,records,rows,receipt,binder):
    verify_native_receipt(receipt)
    if len(records)!=146 or len(rows)!=146 or receipt['cases']!=rows or [r['name'] for r in records]!=[r['case'] for r in rows]:
        raise ValueError('The complete native rows and finite input records differ')
    for expected,row in zip(records,rows):
        if (not all(row.get(k) is True for k in ('coordinates_exactly_unchanged','directed_boundary_preserved',
                'topology_valid','all_triangles_exactly_noncollinear')) or
            row.get('vertices')!=4 or row.get('faces')!=2 or row.get('kept_source_diagonal') is not False or
            row.get('exactly_coplanar')!=expected['expected_exact_coplanarity'] or
            sorted(row['source_diagonal_original_vertex_ids'])!=sorted(expected['original_diagonal_ids']) or
            sorted(row['chosen_diagonal_original_vertex_ids'])!=sorted(set(expected['boundary_ids'])-set(expected['original_diagonal_ids']))):
            raise ValueError('A native quad does not retain its exact input or new diagonal')
    native_packet,binding=binder.bind_operations(packet,rows)
    if binding['cases_verified']!=146 or not binding['normalized_rows_used'] or not binding['plan_face_assignment_preserved']:
        raise ValueError('The native rows are not completely aligned to the finite face assignments')
    return native_packet,binding
