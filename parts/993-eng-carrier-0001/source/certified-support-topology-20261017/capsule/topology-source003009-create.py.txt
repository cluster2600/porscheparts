def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb')as stream:
        for block in iter(lambda:stream.read(1048576),b''):h.update(block)
    return h.hexdigest()


def canonical(value):return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()


def path(name):
    p=Path(name)
    if p.is_absolute()or '..'in p.parts:raise ValueError('Only workspace-relative retained paths are allowed')
    return ROOT/p


def fresh(name):
    p=path(name)
    if p.exists()or any(x.is_symlink()for x in [p,*p.parents]):raise ValueError('A fresh candidate destination is required')
    return p


def private(source,name):
    spec=importlib.util.spec_from_file_location(name,source)
    module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module)
    return module


def write(destination,value):
    with Path(destination).open('x')as stream:json.dump(value,stream,indent=2,allow_nan=False);stream.write('\n')

def read_native_selection(plan):
    selection=private(path(plan['native_selection']),'_real_48PMP_1Euler_selection')
    bundle=json.loads(path(plan['selection_bundle']).read_text())
    actual=selection.assemble(ROOT,json.loads(path(plan['selection_inputs']).read_text()))
    if actual!=bundle:raise ValueError('The selected native topology and its real source receipts differ')
    return bundle


def bind_case_native(packet,rows,binder,expected_quads):
    names=[operation['name']+'-step-'+str(j)for operation in packet['operations']for j,_ in enumerate(operation['steps'])]
    selected=[row for row in rows if row['case']in set(names)]
    if len(names)!=expected_quads or [row['case']for row in selected]!=names:raise ValueError('The native rows for this exact packet differ')
    native,binding=binder.bind_operations(packet,selected)
    if (binding['cases_verified']!=expected_quads or not binding['normalized_rows_used']or
            not binding['plan_face_assignment_preserved']):raise ValueError('The native rows do not bind every planned face assignment')
    return native,binding
