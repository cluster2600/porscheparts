cmake_minimum_required(VERSION 3.15)
find_package(deal.II 9.6 REQUIRED HINTS ${DEAL_II_DIR})
deal_ii_initialize_cached_variables()
project(CarrierOccupation LANGUAGES CXX)
set(ADAMANTINE_SOURCE_DIR "" CACHE PATH "Existing pinned source tree")
set(ADAMANTINE_GEOMETRY_OBJECT "" CACHE FILEPATH "Existing compiled Geometry.cc.o")
if(NOT EXISTS "${ADAMANTINE_SOURCE_DIR}/source/Geometry.hh" OR NOT EXISTS "${ADAMANTINE_GEOMETRY_OBJECT}")
  message(FATAL_ERROR "Existing Adamantine headers and Geometry.cc.o are required")
endif()
add_executable(carrier_occupation adamantine_occupation.cc "${ADAMANTINE_GEOMETRY_OBJECT}")
target_include_directories(carrier_occupation PRIVATE "${ADAMANTINE_SOURCE_DIR}/source")
set_target_properties(carrier_occupation PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
deal_ii_setup_target(carrier_occupation)
