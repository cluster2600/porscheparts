"""Small stdlib/ijson checks only, without scientific runtime imports."""
from array import array
import hashlib
import importlib.util
import io
import json
from pathlib import Path
import struct
import unittest

SOURCE = Path(__file__).with_name('extract_intervals.py')
SPEC = importlib.util.spec_from_file_location('_retained_contact_intervals_check', SOURCE)
m = importlib.util.module_from_spec(SPEC); SPEC.loader.exec_module(m)


class RetainedDisplayChecks(unittest.TestCase):
    def value(self): return dict(numerator='1', denominator='3', nearest_float=1 / 3)
    def test_nearest_float(self): self.assertEqual(m.coordinate(self.value()).hex(), (1 / 3).hex())
    def test_changed_nearest_refused(self):
        with self.assertRaises(ValueError): m.coordinate(dict(self.value(), nearest_float=.3))
    def test_unreduced_fraction_refused(self):
        with self.assertRaises(ValueError): m.coordinate(dict(self.value(), numerator='2', denominator='6'))
    def test_noncanonical_fraction_refused(self):
        with self.assertRaises(ValueError): m.coordinate(dict(self.value(), numerator='01'))
    def test_interval_order_duplicates(self):
        interval = [[self.value(), self.value()], [self.value(), self.value()]]
        values = array('d'); m.append_interval(values, interval); m.append_interval(values, interval)
        self.assertEqual(list(values), [1 / 3] * 8)
    def test_hasher_does_not_consume_hidden_bytes(self):
        raw = b'{"items":[1,2,3]}'; reader = m.HashedReader(io.BytesIO(raw))
        self.assertEqual(list(m.ijson.items(reader, 'items.item')), [1, 2, 3])
        self.assertEqual(reader.digest.hexdigest(), hashlib.sha256(raw).hexdigest())
    def test_npy_little_endian(self):
        raw = m.npy_bytes(array('d', [1., -2.]), (1, 2), '<f8')
        length = struct.unpack('<H', raw[8:10])[0]
        self.assertEqual(raw[:8], b'\x93NUMPY\x01\x00')
        self.assertEqual((10 + length) % 64, 0)
        self.assertIn(b"'shape': (1, 2)", raw[10:10 + length])
        self.assertEqual(struct.unpack('<2d', raw[10 + length:]), (1., -2.))
    def test_wrong_npy_shape_refused(self):
        with self.assertRaises(ValueError): m.npy_bytes(array('q', [1]), (2,), '<i8')


if __name__ == '__main__': unittest.main()
