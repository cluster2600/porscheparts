cmake_minimum_required(VERSION 3.15)
find_package(deal.II 9.6 REQUIRED HINTS ${DEAL_II_DIR})
deal_ii_initialize_cached_variables()
project(AdamantineExporterRegression LANGUAGES CXX)
set(ADAMANTINE_SOURCE_DIR "" CACHE PATH "External Adamantine source tree to test")
if(NOT ADAMANTINE_SOURCE_DIR OR NOT EXISTS "${ADAMANTINE_SOURCE_DIR}/source/PostProcessor.cc")
  message(FATAL_ERROR "Set ADAMANTINE_SOURCE_DIR to an external Adamantine source tree")
endif()
add_executable(export_regression tests/test_native_export_regression.cc "${ADAMANTINE_SOURCE_DIR}/source/PostProcessor.cc")
target_include_directories(export_regression PRIVATE "${ADAMANTINE_SOURCE_DIR}/source")
set_target_properties(export_regression PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
deal_ii_setup_target(export_regression)
enable_testing()
add_test(NAME native_export COMMAND export_regression ${CMAKE_CURRENT_BINARY_DIR}/ctest-output)
set_tests_properties(native_export PROPERTIES TIMEOUT 60)
